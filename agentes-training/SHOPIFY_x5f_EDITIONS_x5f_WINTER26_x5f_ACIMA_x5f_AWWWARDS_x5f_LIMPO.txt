JS Lenis parallax limpo (sem deformar):
import Lenis from 'lenis'

const lenis = new Lenis({ duration: 1.2 })

lenis.on('scroll', ({ scroll })=>{
  document.querySelectorAll('[data-depth]').forEach(layer=>{
    const depth = parseFloat(layer.dataset.depth)
    const y = scroll * depth * 0.3 // 0.3 = sutil, não 2x que fica enjoado
    layer.style.transform = `translate3d(0, ${-y}px, 0) translateZ(${depth * 100}px)`
  })
})

function raf(t){
  lenis.raf(t)
  requestAnimationFrame(raf)
}

TEMPLATE 2: LUZ RENASCENTISTA - SHADER QUE NÃO É OBJETO
================================================================================

Shopify Winter 26 não tem toro. Tem luz que parece pintura renascentista.

Shader único canvas full-screen, atrás do conteúdo, com luz que segue scroll:

Canvas:
<canvas id="renaissance-light" style="position: fixed; inset: 0; z-index: -1; pointer-events: none;"></canvas>

Shader (OGL, 12kB, não Three.js pesado):
import { Renderer, Program, Mesh, Triangle } from 'ogl'

const renderer = new Renderer({ dpr: Math.min(2, window.devicePixelRatio) })
const gl = renderer.gl
document.getElementById('renaissance-light').appendChild(gl.canvas)

const geometry = new Triangle(gl)

const program = new Program(gl, {
  vertex: `attribute vec2 uv; attribute vec2 position; varying vec2 vUv; void main(){ vUv = uv; gl_Position = vec4(position, 0, 1); }`,
  fragment: `
    precision highp float;
    varying vec2 vUv;
    uniform float uTime;
    uniform float uScroll;
    uniform vec2 uPointer;

    // Luz renascentista: 2 pontos de luz quente, como janela de igreja
    void main(){
      vec2 st = vUv;
      st.x -= 0.5;
      st.x *= 1.5; // aspect
      st += 0.5;

      // Luz 1: topo, quente ouro
      float d1 = distance(st, vec2(0.5 + sin(uTime * 0.1) * 0.05, 0.2));
      float light1 = 1.0 - smoothstep(0.0, 0.8, d1);
      light1 *= 0.15;

      // Luz 2: lateral, fria
      float d2 = distance(st, vec2(0.8, 0.6 + cos(uTime * 0.08) * 0.05));
      float light2 = 1.0 - smoothstep(0.0, 0.6, d2);
      light2 *= 0.08;

      // Vignette renascentista
      float vignette = 1.0 - smoothstep(0.3, 1.2, distance(st, vec2(0.5)));

      vec3 color = vec3(0.02, 0.02, 0.03); // obsidian quase preto
      color += vec3(0.8, 0.65, 0.35) * light1; // gold quente
      color += vec3(0.4, 0.5, 0.9) * light2; // azul frio
      color *= vignette;

      // Grain sutil renascentista
      float grain = fract(sin(dot(st, vec2(12.9898,78.233))) * 43758.5453) * 0.02;

      gl_FragColor = vec4(color + grain, 1.0);
    }
  `,
  uniforms: {
    uTime: { value: 0 },
    uScroll: { value: 0 },
    uPointer: { value: [0.5,0.5] }
  }
})

const mesh = new Mesh(gl, { geometry, program })

function raf(t){
  requestAnimationFrame(raf)
  program.uniforms.uTime.value = t * 0.001
  renderer.render({ scene: mesh })
}
raf()

Lenis sync:
lenis.on('scroll', ({ progress })=>{
  program.uniforms.uScroll.value = progress
})

Pointer sutil:
window.addEventListener('pointermove', (e)=>{
  program.uniforms.uPointer.value = [e.clientX / innerWidth, 1.0 - e.clientY / innerHeight]
})

GSAP reveal com ScrollTrigger (sem objeto 3D feio):
import gsap from 'gsap'
import ScrollTrigger from 'gsap/ScrollTrigger'

gsap.registerPlugin(ScrollTrigger)

document.querySelectorAll('.chapter').forEach(chapter=>{
  const title = chapter.querySelector('.chapter-title')
  const desc = chapter.querySelector('.chapter-desc')
  const aura = chapter.querySelector('.chapter-aura')
  const kicker = chapter.querySelector('.chapter-kicker')

  const tl = gsap.timeline({
    scrollTrigger: {
      trigger: chapter,
      start: 'top 70%',
      end: 'bottom 30%',
      scrub: 1
    }
  })

  tl.fromTo(kicker, { y: 20, opacity: 0 }, { y: 0, opacity: 0.5, duration: 0.5 }, 0)
    .fromTo(title, { y: 100, opacity: 0 }, { y: 0, opacity: 1, duration: 1, ease: 'power3.out' }, 0.1)
    .fromTo(desc, { y: 30, opacity: 0 }, { y: 0, opacity: 0.7, duration: 0.8 }, 0.3)
    .fromTo(aura, { scale: 0.8, opacity: 0 }, { scale: 1.1, opacity: 0.15, duration: 1.5 }, 0)
})

Por que acima: Texto é hero, não objeto. Aura é luz, não mesh. Scroll é scrub 1 (lento, editorial), não 0 (rápido, enjoado).
